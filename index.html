headers.forEach((h, ci) => { if (!h) return; let val = row[ci]; if (val instanceof Date) val = Utilities.formatDate(val, Session.getScriptTimeZone(), "dd/MM/yyyy"); obj[h] = val; if (val !== "" && val !== null && val !== undefined) temValor = true; }); if (temValor) rows.push(obj); } result.movimentacoes = rows; return; } // ── abas CD_tab* ─────────────────────────────────────────── if (!nome.startsWith("CD_tab")) return; const tabKey = nome.replace("CD_", ""); return; } const ss = SpreadsheetApp.getActiveSpreadsheet(); const config = ss.getSheetByName("Config") || ss.getSheetByName("config"); if (!config) { SpreadsheetApp.getUi().alert("Aba 'Config' não encontrada."); const urlParam = link.replace("https://www.comdinheiro.com.br/", ""); const payload = `username=${encodeURIComponent(user)}&password=${encodeURIComponent(pass)}&URL=${encodeURIComponent(urlParam)}&format=json3`; }); const criadas = []; nomes.forEach(tabName => { const tabObj = tables[tabName]; const sheet = ss.insertSheet("CD_" + tabName); sheet.setTabColor("#0f3d57"); if (!tabObj || typeof tabObj !== "object") { sheet.getRange("A1").setValue("(vazia)"); sheet.setFrozenRows(1); sheet.autoResizeColumns(1, maxCols); const info = `CD_${tabName}: ${linhas.length} linhas × ${maxCols} colunas`; criadas.push(info); Logger.log("✅ " + info); }); SpreadsheetApp.getUi().alert("✅ " + criadas.length + " abas CD_tab* criadas.\n\n" + criadas.join("\n")); } // ── DUMP MOVIMENTAÇÕES → aba movimentacoes ──────────────────────── function dumpMovimentacoes() { const props = PropertiesService.getScriptProperties(); const user = props.getProperty("COMDINHEIRO_USER"); const payload = `username=${encodeURIComponent(user)}&password=${encodeURIComponent(pass)}&URL=${encodeURIComponent(urlParam)}&format=json3`; const response = UrlFetchApp.fetch(CD_ENDPOINT, { method: "post", contentType: "application/x-www-form-urlencoded", payload, muteHttpExceptions: true }); sheet.getRange(2, 1, rows.length-1, 1).setNumberFormat("dd/mm/yyyy"); const n = rows.length - 1; Logger.log("✅ movimentacoes → " + n + " registros"); SpreadsheetApp.getUi().alert("✅ " + n + " movimentações gravadas.");
